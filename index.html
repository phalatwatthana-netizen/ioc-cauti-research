  <script>

    // --- STATE SETTINGS ---

    let MOCK_EXPERTS = [];

    let MOCK_DATA = { phases: [], tools: [], items: [] };


    const appState = {

      currentUser: null,

      isAdmin: false,

      view: 'dashboard', 

      selectedTool: null,

      adminTab: 'overview',

      evaluations: {}, 

      toolStatuses: {},

    };


    const app = {

      init() {

        document.getElementById('app-container').innerHTML = '<div class="flex justify-center items-center h-64 flex-col gap-4"><div class="loader"></div><div class="text-teal-600 font-bold text-lg animate-pulse">กำลังโหลดข้อมูลจาก Google Sheets...</div></div>';

        

        // Call Google Apps Script function

        if (typeof google !== 'undefined') {

          google.script.run

            .withSuccessHandler((serverData) => {

              MOCK_EXPERTS = serverData.experts;

              MOCK_DATA.phases = serverData.phases;

              MOCK_DATA.tools = serverData.tools;

              MOCK_DATA.items = serverData.items;

              appState.evaluations = serverData.evaluations || {};

              appState.toolStatuses = serverData.toolStatuses || {};

              

              if(MOCK_EXPERTS.length > 0) {

                appState.currentUser = MOCK_EXPERTS[0];

              }

              

              this.populateRoleSelector();

              this.render();

            })

            .withFailureHandler((error) => {

              document.getElementById('app-container').innerHTML = `<div class="text-red-500 p-8 text-center bg-red-50 rounded-xl border border-red-200"><i class="fa-solid fa-triangle-exclamation text-3xl mb-3"></i><br>ไม่สามารถโหลดข้อมูลได้: ${error.message}</div>`;

            })

            .getAppData();

        } else {

           document.getElementById('app-container').innerHTML = `<div class="text-orange-500 p-8 text-center bg-orange-50 rounded-xl">ไม่พบสภาพแวดล้อม Google Apps Script กรุณารันผ่านการ Deploy ของ GAS</div>`;

        }

      },


      getTemplate(id) { return document.getElementById(id).content.cloneNode(true); },

      clearContainer(id) { document.getElementById(id).innerHTML = ''; },


      showToast(type, message) {

        const container = document.getElementById('toast-container');

        const toast = document.createElement('div');

        const bg = type === 'error' ? 'bg-red-500' : 'bg-teal-600';

        const icon = type === 'error' ? 'fa-circle-exclamation' : 'fa-circle-check';

        

        toast.className = `${bg} text-white px-6 py-3 rounded-lg shadow-lg flex items-center gap-2 transform transition-all duration-300 translate-y-full opacity-0`;

        toast.innerHTML = `<i class="fa-solid ${icon}"></i> <span class="font-medium">${message}</span>`;

        container.appendChild(toast);

        

        setTimeout(() => toast.classList.remove('translate-y-full', 'opacity-0'), 10);

        setTimeout(() => {

          toast.classList.add('opacity-0');

          setTimeout(() => toast.remove(), 300);

        }, 3000);

      },


      getStatusBadgeHTML(status) {

        if (status === 'submitted') return `<span class="inline-flex items-center gap-1 px-2.5 py-1 rounded-full text-xs font-medium bg-green-100 text-green-800 border border-green-500"><i class="fa-solid fa-circle-check"></i> ประเมินแล้ว</span>`;

        if (status === 'draft') return `<span class="inline-flex items-center gap-1 px-2.5 py-1 rounded-full text-xs font-medium bg-yellow-100 text-yellow-800 border border-yellow-500"><i class="fa-solid fa-clock"></i> บันทึกร่าง</span>`;

        return `<span class="inline-flex items-center gap-1 px-2.5 py-1 rounded-full text-xs font-medium bg-rose-100 text-rose-800 border border-rose-500"><i class="fa-solid fa-circle-exclamation"></i> ยังไม่ประเมิน</span>`;

      },


      switchRole(value) {

        if (value === 'admin') {

          appState.isAdmin = true;

        } else {

          appState.isAdmin = false;

          appState.currentUser = MOCK_EXPERTS.find(exp => exp.id === value);

          appState.view = 'dashboard';

        }

        this.render();

      },


      setView(viewName) {

        appState.view = viewName;

        window.scrollTo(0, 0);

        this.render();

      },


      openEvaluation(toolId) {

        appState.selectedTool = MOCK_DATA.tools.find(t => t.id === toolId);

        this.setView('evaluate');

      },


      setAdminTab(tabName) {

        appState.adminTab = tabName;

        this.renderAdminTabContent();

        this.updateAdminTabStyles();

      },


      handleScoreChange(itemId, score) {

        const key = `${appState.currentUser.id}_${appState.selectedTool.id}_${itemId}`;

        if (!appState.evaluations[key]) appState.evaluations[key] = {};

        appState.evaluations[key].score = score;

      },


      handleCommentChange(itemId, comment) {

        const key = `${appState.currentUser.id}_${appState.selectedTool.id}_${itemId}`;

        if (!appState.evaluations[key]) appState.evaluations[key] = {};

        appState.evaluations[key].comment = comment;

      },


      prepareSavePayload() {

        const uid = appState.currentUser.id;

        const tool = appState.selectedTool;

        const items = MOCK_DATA.items.filter(i => i.toolId === tool.id);

        let evalsToSave = [];

        

        items.forEach(i => {

           const val = appState.evaluations[`${uid}_${tool.id}_${i.id}`];

           if(val && val.score !== undefined) {

              evalsToSave.push({ itemId: i.id, score: val.score, comment: val.comment || "" });

           }

        });

        return evalsToSave;

      },


      saveDraft(event) {

        const uid = appState.currentUser.id;

        const tid = appState.selectedTool.id;

        const currentStatus = appState.toolStatuses[uid]?.[tid];

        const newStatus = currentStatus === 'submitted' ? 'submitted' : 'draft';

        const payload = this.prepareSavePayload();

        

        const btn = event.currentTarget;

        const originalHtml = btn.innerHTML;

        btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> กำลังบันทึก...';

        btn.disabled = true;


        google.script.run

          .withSuccessHandler(() => {

            if (!appState.toolStatuses[uid]) appState.toolStatuses[uid] = {};

            appState.toolStatuses[uid][tid] = newStatus;

            

            this.showToast('success', 'บันทึกร่างลงฐานข้อมูลเรียบร้อยแล้ว');

            btn.innerHTML = originalHtml;

            btn.disabled = false;

            

            document.getElementById('eval-status-container').innerHTML = 

              this.getStatusBadgeHTML(newStatus) + 

              (newStatus === 'submitted' ? `<span class="text-sm text-slate-500 flex items-center gap-1 ml-2"><i class="fa-solid fa-circle-info"></i> สามารถแก้ไขและอัปเดตผลได้</span>` : '');

          })

          .withFailureHandler(() => {

            this.showToast('error', 'เกิดข้อผิดพลาดในการบันทึกร่าง');

            btn.innerHTML = originalHtml;

            btn.disabled = false;

          })

          .saveEvaluationData(uid, tid, newStatus, payload);

      },


      submitEvaluation(event) {

        const uid = appState.currentUser.id;

        const tool = appState.selectedTool;

        const items = MOCK_DATA.items.filter(i => i.toolId === tool.id);

        

        const isComplete = items.every(i => {

          const val = appState.evaluations[`${uid}_${tool.id}_${i.id}`];

          return val && val.score !== undefined;

        });


        if (!isComplete) {
